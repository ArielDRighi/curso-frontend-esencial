💻 Ejemplo de código Go que verás en el curso:

package main

import (
    "fmt"
    "time"
    "sync"
)

func main() {
    var wg sync.WaitGroup
    
    // Demostrar concurrencia con goroutines
    for i := 1; i <= 3; i++ {
        wg.Add(1)
        go func(id int) {
            defer wg.Done()
            fmt.Printf("🚀 Goroutine %d ejecutándose\n", id)
            time.Sleep(time.Second)
            fmt.Printf("✅ Goroutine %d completada\n", id)
        }(i)
    }
    
    wg.Wait()
    fmt.Println("🎉 ¡Todas las goroutines completadas!")
}
